# secp256k1 JNI (ACINQ kotlin-secp256k1) — the NOSTR layer uses it for Schnorr sign/verify and
# ECDH (NIP-44/NIP-17 + NOSTR identity derivation). JNI binds native methods by fully-qualified
# class name, so R8 must NOT rename/strip these classes or the native binding fails at runtime with
# "Could not load native Secp256k1 JNI library" — which crashes wallet creation at
# NOSTRIdentity.fromSeed (upstream Zcash has no secp256k1 dep, so upstream rules don't cover it).
-keep class fr.acinq.secp256k1.** { *; }
-keepclasseswithmembernames class fr.acinq.secp256k1.** { native <methods>; }
-dontwarn fr.acinq.secp256k1.**

# WebRTC (io.getstream:stream-webrtc-android) — voice/video calls. Binds native methods in
# org.webrtc.* by class name; R8 renaming breaks the JNI at call time. Not on the launch path
# (calls-only), but kept so a call doesn't crash the app.
-keep class org.webrtc.** { *; }
-keepclasseswithmembernames class org.webrtc.** { native <methods>; }
-dontwarn org.webrtc.**

# zxing-cpp native QR decoder (libzxingcpp_android.so) — camera QR scanning. Same JNI-by-name
# concern; kept so the scanner doesn't crash under R8.
-keep class zxingcpp.** { *; }
-keep class com.zxingcpp.** { *; }
-dontwarn zxingcpp.**
-dontwarn com.zxingcpp.**

# SECURITY (privacy): strip android.util.Log.{v,d,i,w} from optimized release builds. ZCHAT debug
# logs carry conversation IDs, sender hashes, file URLs and addresses; even with per-call redaction,
# removing the calls entirely in release eliminates the on-device logcat side channel. Requires the
# -optimize proguard base (proguard-android-optimize.txt, already used above). Log.e/wtf are kept so
# genuine error diagnostics survive. The string-building args become dead code and are removed too.
-assumenosideeffects class android.util.Log {
    public static int v(...);
    public static int d(...);
    public static int i(...);
    public static int w(...);
    public static boolean isLoggable(...);
}
